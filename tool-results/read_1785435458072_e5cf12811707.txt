    73→    73→// Formata data brasileira: 10042004 → 10/04/2004
    74→    74→function formatarData(valor: string): string {
    75→    75→  // Remove tudo que não é dígito
    76→    76→  const digitos = valor.replace(/\D/g, "");
    77→    77→  
    78→    78→  // Limita a 8 dígitos (DDMMYYYY)
    79→    79→  const limitados = digitos.slice(0, 8);
    80→    80→  
    81→    81→  // Aplica a máscara progressiva
    82→    82→  if (limitados.length === 0) return "";
    83→    83→  if (limitados.length <= 2) return limitados;
    84→    84→  if (limitados.length <= 4) return `${limitados.slice(0, 2)}/${limitados.slice(2)}`;
    85→    85→  return `${limitados.slice(0, 2)}/${limitados.slice(2, 4)}/${limitados.slice(4)}`;
    86→    86→}
