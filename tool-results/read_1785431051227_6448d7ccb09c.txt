    70→// Tree node component for visualization
    71→function TreeNode({ 
    72→  no, 
    73→  onEdit, 
    74→  depth = 0 
    75→}: { 
    76→  no: NoArvore; 
    77→  onEdit: (no: NoArvore) => void;
    78→  depth?: number;
    79→}) {
    80→  const relacaoLabels: Record<string, string> = {
    81→    requerente: "Requerente",
    82→    pai: "Pai",
    83→    mae: "Mãe",
    84→    avo_paterno: "Avô Paterno",
    85→    avo_paterna: "Avó Paterna",
    86→    avo_materno: "Avô Materno",
    87→    avo_materna: "Avó Materna",
    88→    ancestral: "Ancestral Europeu",
    89→  };
    90→
    91→  return (
    92→    <div 
    93→      className="group relative"
    94→      style={{ marginLeft: depth > 0 ? `${depth * 24}px` : 0 }}
    95→    >
    96→      <Card className={`transition-all hover:shadow-md cursor-pointer ${
    97→        no.relacao === 'ancestral' ? 'border-gold/40 bg-gold/5' : 'border-navy/10'
    98→      }`}>
    99→        <CardContent className="p-3 sm:p-4" onClick={() => onEdit(no)}>
   100→          <div className="flex items-start gap-3">
   101→            {/* Icon */}
   102→            <div className={`size-10 rounded-lg flex items-center justify-center shrink-0 ${
   103→              no.relacao === 'ancestral' ? 'bg-gold/20' : 'bg-navy/10'
   104→            }`}>
   105→              {no.relacao === 'requerente' || no.relacao === 'ancestral' ? (
   106→                <User className={`size-5 ${no.relacao === 'ancestral' ? 'text-gold-dark' : 'text-navy'}`} />
   107→              ) : (
   108→                <Users className="size-5 text-navy/70" />
   109→              )}
   110→            </div>
   111→
   112→            {/* Info */}
   113→            <div className="flex-1 min-w-0">
   114→              <div className="flex items-center gap-2 flex-wrap">
   115→                <h4 className="font-medium text-navy truncate">{no.nomeCompleto}</h4>
   116→                <Badge variant="outline" className="text-[10px]">
   117→                  {relacaoLabels[no.relacao] || no.relacao}
   118→                </Badge>
   119→                <Badge className={`${getStatusRegistroColor(no.statusRegistro)} text-[10px]`}>
   120→                  {getStatusRegistroLabel(no.statusRegistro)}
   121→                </Badge>
   122→              </div>
   123→              
   124→              {(no.nascimento || no.casamento || no.obito) && (
   125→                <div className="flex flex-wrap gap-2 mt-1.5 text-[11px] text-muted-foreground">
   126→                  {no.nascimento && (
   127→                    <span className="flex items-center gap-1">
   128→                      <CalendarDays className="size-3" />
   129→                      Nasc: {no.nascimento.data}{no.nascimento.local ? ` - ${no.nascimento.local}` : ''}
   130→                    </span>
   131→                  )}
   132→                  {no.casamento && (
   133→                    <span className="flex items-center gap-1">
   134→                      <FileText className="size-3" />
   135→                      Casam: {no.casamento.data}
   136→                    </span>
   137→                  )}
   138→                  {no.obito && (
   139→                    <span className="flex items-center gap-1">
   140→                      <AlertCircle className="size-3" />
   141→                      Óbito: {no.obito.data}
   142→                    </span>
   143→                  )}
   144→                </div>
   145→              )}
   146→              
   147→              {no.variacoesGrafia && (
   148→                <p className="mt-1 text-[11px] text-muted-foreground italic">
   149→                  Var.: {no.variacoesGrafia}
   150→                </p>
   151→              )}
   152→            </div>
   153→
   154→            {/* Edit indicator */}
   155→            <Edit2 className="size-4 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity shrink-0" />
   156→          </div>
   157→        </CardContent>
   158→      </Card>
   159→
   160→      {/* Connector line */}
   161→      {depth < 3 && (
   162→        <div className="w-px h-4 bg-navy/20 ml-5" />
   163→      )}
   164→    </div>
   165→  );
   166→}

   168→export default function FichaCliente() {
   169→  const router = useRouter();
   170→  const searchParams = useSearchParams();
   171→  const clienteId = searchParams.get("cliente");
   172→  
   173→  const [cliente, setCliente] = useState<Cliente | null>(null);
   174→  const [abaAtiva, setAbaAtiva] = useState("arvore");
   175→  const [salvando, setSalvando] = useState(false);
   176→  
   177→  // Modal states
   178→  const [noSelecionado, setNoSelecionado] = useState<NoArvore | null>(null);
   179→  const [dialogNoAberto, setDialogNoAberto] = useState(false);
   180→  const [dialogNovoNoAberto, setDialogNovoNoAberto] = useState(false);
   181→  const [dialogDespesaAberto, setDialogDespesaAberto] = useState(false);
   182→  
   183→  // Form states
   184→  const [novoNoRelacao, setNovoNoRelacao] = useState("pai");
   185→  const [formNo, setFormNo] = useState<Partial<NoArvore>>({});
   186→  const [formDespesa, setFormDespesa] = useState({
   187→    descricao: "",
   188→    valor: "",
   189→    moeda: "BRL" as "BRL" | "EUR" | "USD",
   190→    categoria: "outro" as Despesa["categoria"],
   191→  });
   192→  const [novaEntradaTitulo, setNovaEntradaTitulo] = useState("");
   193→  const [novaEntradaDescricao, setNovaEntradaDescricao] = useState("");
   194→  const [novoLink, setNovoLink] = useState("");
   195→
   196→  // Load client data
   197→  function carregarCliente() {
   198→    if (clienteId) {
   199→      const c = getClienteById(clienteId);
   200→      if (c) {
   201→        setCliente(c);
   202→      } else {
   203→        router.push("/painel");
   204→      }
   205→    }
   206→  }
   207→
   208→  useEffect(() => {
   209→    carregarCliente();
   210→  }, [clienteId]);
   211→
   212→  async function handleSalvar() {
   213→    if (!cliente) return;
   214→    
   215→    setSalvando(true);
   216→    await new Promise(r => setTimeout(r, 500));
   217→    
   218→    updateCliente(cliente.id, cliente);
   219→    setSalvando(false);
   220→    
   221→    // Show success feedback
   222→    alert("Ficha salva com sucesso!");
   223→  }
   224→
   225→  function handleStatusChange(novoStatus: StatusCliente) {
   226→    if (!cliente) return;
   227→    setCliente({ ...cliente, status: novoStatus });
   228→  }
   229→
   230→  // Tree operations
   231→  function abrirEditarNo(no: NoArvore) {
   232→    setFormNo({ ...no });
   233→    setNoSelecionado(no);
   234→    setDialogNoAberto(true);
   235→  }
   236→
   237→  function salvarNo() {
   238→    if (!cliente || !noSelecionado) return;
   239→    
   240→    const atualizado = updateNoArvore(cliente.id, noSelecionado.id, formNo);
   241→    if (atualizado) {
   242→      setCliente({
   243→        ...cliente,
   244→        arvore: cliente.arvore.map(n => n.id === noSelecionado.id ? atualizado! : n)
   245→      });
   246→    }
   247→    
   248→    setDialogNoAberto(false);
   249→    setNoSelecionado(null);
   250→    setFormNo({});
   251→  }
   252→
   253→  function adicionarNovoNo() {
   254→    if (!cliente) return;
   255→    
   256→    const novoNo = addNoArvore(cliente.id, {
   257→      nomeCompleto: "",
   258→      relacao: novoNoRelacao as NoArvore["relacao"],
   259→      statusRegistro: "pendente",
   260→      variacoesGrafia: "",
   261→      anotacoesCartorio: "",
   262→    });
   263→    
   264→    if (novoNo) {
   265→      setCliente({
   266→        ...cliente,
   267→        arvore: [...cliente.arvore, novoNo]
   268→      });
   269→      
   270→      // Open edit dialog for the new node
   271→      setFormNo({ ...novoNo });
   272→      setNoSelecionado(novoNo);
   273→      setDialogNoAberto(true);
   274→    }
   275→    
   276→    setDialogNovoNoAberto(false);
   277→  }
   278→
   279→  function excluirNo(noId: string) {
   280→    if (!cliente) return;
   281→    
   282→    deleteNoArvore(cliente.id, noId);
   283→    setCliente({
   284→      ...cliente,
   285→      arvore: cliente.arvore.filter(n => n.id !== noId)
   286→    });
   287→  }
   288→
   289→  // Financial operations
   290→  function adicionarDespesa() {
   291→    if (!cliente) return;
   292→    
   293→    const valorNumerico = parseFloat(formDespesa.valor.replace(",", "."));
   294→    if (isNaN(valorNumerico)) {
   295→      alert("Valor inválido");
   296→      return;
   297→    }
   298→    
   299→    const novaDespesa = addDespesa(cliente.id, {
   300→      descricao: formDespesa.descricao,
   301→      valor: valorNumerico,
   302→      moeda: formDespesa.moeda,
   303→      data: new Date().toLocaleDateString("pt-BR"),
   304→      categoria: formDespesa.categoria,
   305→    });
   306→    
   307→    if (novaDespesa) {
   308→      setCliente({
   309→        ...cliente,
   310→        financeiro: {
   311→          ...cliente.financeiro,
   312→          despesas: [...cliente.financeiro.despesas, novaDespesa]
   313→        }
   314→      });
   315→    }
   316→    
   317→    setDialogDespesaAberto(false);
   318→    setFormDespesa({ descricao: "", valor: "", moeda: "BRL", categoria: "outro" });
   319→  }
   320→
   321→  function excluirDespesa(despesaId: string) {
   322→    if (!cliente) return;
   323→    
   324→    deleteDespesa(cliente.id, despesaId);
   325→    setCliente({
   326→      ...cliente,
   327→      financeiro: {
   328→        ...cliente.financeiro,
   329→        despesas: cliente.financeiro.despesas.filter(d => d.id !== despesaId)
   330→      }
   331→    });
   332→  }
   333→
   334→  function alterarStatusPagamento(tipo: "taxaAnalise" | "taxaSucesso", status: "pago" | "pendente" | "aguardando") {
   335→    if (!cliente) return;
   336→    
   337→    updateStatusPagamento(cliente.id, tipo, status);
   338→    
   339→    setCliente({
   340→      ...cliente,
   341→      financeiro: {
   342→        ...cliente.financeiro,
   343→        [tipo]: {
   344→          ...cliente.financeiro[tipo],
   345→          status
   346→        }
   347→      }
   348→    });
   349→  }
   350→
   351→  // Timeline operations
   352→  function adicionarEntradaTimeline() {
   353→    if (!cliente || !novaEntradaTitulo.trim()) return;
   354→    
   355→    const links = novoLink.trim() ? [novoLink.trim()] : undefined;
   356→    
   357→    const novaEntrada = addEntradaTimeline(cliente.id, {
   358→      data: new Date().toLocaleDateString("pt-BR"),
   359→      titulo: novaEntradaTitulo.trim(),
   360→      descricao: novaEntradaDescricao.trim(),
   361→      links,
   362→    });
   363→    
   364→    if (novaEntrada) {
   365→      setCliente({
   366→        ...cliente,
   367→        timeline: [novaEntrada, ...cliente.timeline]
   368→      });
   369→    }
   370→    
   371→    setNovaEntradaTitulo("");
   372→    setNovaEntradaDescricao("");
   373→    setNovoLink("");
   374→  }
   375→
   376→  function excluirEntrada(entradaId: string) {
   377→    if (!cliente) return;
   378→    
   379→    deleteEntradaTimeline(cliente.id, entradaId);
   380→    setCliente({
   381→      ...cliente,
   382→      timeline: cliente.timeline.filter(e => e.id !== entradaId)
   383→    });
   384→  }
   385→
   386→  if (!cliente) {
   387→    return (
   388→      <div className="flex items-center justify-center h-64">
   389→        <div className="size-8 border-3 border-navy/20 border-t-navy rounded-full animate-spin" />
   390→      </div>
   391→    );
   392→  }
   393→
   394→  const lucroLiquido = calcularLucroLiquido(cliente);
   395→
   396→  return (
   397→    <div className="space-y-6">
   398→      {/* Header */}
   399→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   400→        <div className="flex items-center gap-3">
   401→          <Button
   402→            variant="ghost"
   403→            size="icon"
   404→            onClick={() => router.push("/painel")}
   405→            className="text-navy"
   406→          >
   407→            <ArrowLeft className="size-5" />
   408→          </Button>
   409→          
   410→          <div>
   411→            <h1 className="font-serif text-xl font-700 text-navy sm:text-2xl">
   412→              {cliente.nome}
   413→            </h1>
   414→            <p className="text-sm text-muted-foreground">
   415→              Cliente desde {cliente.dataEntrada} • ID: {cliente.id.slice(-6).toUpperCase()}
   416→            </p>
   417→          </div>
   418→        </div>
   419→
   420→        <div className="flex items-center gap-2">
   421→          {/* Status selector */}
   422→          <Select value={cliente.status} onValueChange={(v) => handleStatusChange(v as StatusCliente)}>
   423→            <SelectTrigger className="w-44 border-navy/20">
   424→              <SelectValue />
   425→            </SelectTrigger>
   426→            <SelectContent>
   427→              <SelectItem value="em_analise">Em Análise</SelectItem>
   428→              <SelectItem value="pesquisa_europa">Pesquisa na Europa</SelectItem>
   429→              <SelectItem value="aguardando_emissao">Aguardando Emissão</SelectItem>
   430→              <SelectItem value="concluido">Concluído</SelectItem>
   431→              <SelectItem value="cancelado">Cancelado</SelectItem>
   432→            </SelectContent>
   433→          </Select>
   434→
   435→          <Button
   436→            onClick={handleSalvar}
   437→            disabled={salvando}
   438→            className="bg-navy hover:bg-navy-light text-white"
   439→          >
   440→            {salvando ? (
   441→              <span className="size-4 border-2 border-white/30 border-t-white rounded-full animate-spin mr-2" />
   442→            ) : (
   443→              <Save className="size-4 mr-2" />
   444→            )}
   445→            Salvar
   446→          </Button>
   447→        </div>
   448→      </div>
   449→
   450→      {/* Tabs */}
   451→      <Tabs value={abaAtiva} onValueChange={setAbaAtiva}>
   452→        <TabsList className="bg-navy/5 border border-border w-full justify-start overflow-x-auto">
   453→          <TabsTrigger value="arvore" className="data-[state=active]:bg-navy data-[state=active]:text-white">
   454→            <TreePine className="size-4 mr-2" />
   455→            Árvore Genealógica
   456→          </TabsTrigger>
   457→          <TabsTrigger value="financeiro" className="data-[state=active]:bg-navy data-[state=active]:text-white">
   458→            <DollarSign className="size-4 mr-2" />
   459→            Financeiro
   460→          </TabsTrigger>
   461→          <TabsTrigger value="timeline" className="data-[state=active]:bg-navy data-[state=active]:text-white">
   462→            <Clock className="size-4 mr-2" />
   463→            Diário de Pesquisa
   464→          </TabsTrigger>
   465→        </TabsList>
   466→
   467→        {/* TAB 1: Árvore Genealógica */}
   468→        <TabsContent value="arvore" className="mt-6 space-y-6">
   469→          <div className="flex items-center justify-between">
   470→            <div>
   471→              <h2 className="font-serif text-lg font-600 text-navy">
   472→                Linhagem Familiar & Documentos
   473→              </h2>
   474→              <p className="text-sm text-muted-foreground mt-1">
   475→                Clique em qualquer nó da árvore para editar os dados. Adicione membros da família para mapear a linhagem completa.
   476→              </p>
   477→            </div>
   478→            
   479→            <Dialog open={dialogNovoNoAberto} onOpenChange={setDialogNovoNoAberto}>
   480→              <DialogTrigger asChild>
   481→                <Button variant="outline" className="border-navy/30 text-navy hover:bg-navy hover:text-white">
   482→                  <Plus className="size-4 mr-2" />
   483→                  Adicionar Membro
   484→                </Button>
   485→              </DialogTrigger>
   486→              <DialogContent>
   487→                <DialogHeader>
   488→                  <DialogTitle className="font-serif text-navy">Adicionar Membro da Família</DialogTitle>
   489→                  <DialogDescription>
   490→                    Selecione o grau de parentesco do novo membro.
   491→                  </DialogDescription>
   492→                </DialogHeader>
   493→                
   494→                <div className="space-y-4 mt-4">
   495→                  <div className="space-y-2">
   496→                    <label className="text-sm font-medium text-navy">Grau de Parentesco *</label>
   497→                    <Select value={novoNoRelacao} onValueChange={setNovoNoRelacao}>
   498→                      <SelectTrigger>
   499→                        <SelectValue />
   500→                      </SelectTrigger>
   501→                      <SelectContent>
   502→                        <SelectItem value="pai">Pai</SelectItem>
   503→                        <SelectItem value="mae">Mãe</SelectItem>
   504→                        <SelectItem value="avo_paterno">Avô Paterno</SelectItem>
   505→                        <SelectItem value="avo_paterna">Avó Paterna</SelectItem>
   506→                        <SelectItem value="avo_materno">Avô Materno</SelectItem>
   507→                        <SelectItem value="avo_materna">Avó Materna</SelectItem>
   508→                        <SelectItem value="ancestral">Ancestral Europeu / Emigrante</SelectItem>
   509→                      </SelectContent>
   510→                    </Select>
   511→                  </div>
   512→
   513→                  <div className="flex justify-end gap-2 pt-2">
   514→                    <Button variant="outline" onClick={() => setDialogNovoNoAberto(false)}>
   515→                      Cancelar
   516→                    </Button>
   517→                    <Button onClick={adicionarNovoNo} className="bg-navy hover:bg-navy-light text-white">
   518→                      Adicionar
   519→                    </Button>
   520→                  </div>
   521→                </div>
   522→              </DialogContent>
   523→            </Dialog>
   524→          </div>
   525→
   526→          {/* Tree visualization */}
   527→          <div className="space-y-3">
   528→            {cliente.arvore.length === 0 ? (
   529→              <Card className="border-dashed border-2">
   530→                <CardContent className="py-12 text-center">
   531→                  <TreePine className="size-12 mx-auto text-muted-foreground/30 mb-4" />
   532→                  <h3 className="font-serif text-lg font-600 text-navy mb-2">
   533→                    Árvore Vazia
   534→                  </h3>
   535→                  <p className="text-sm text-muted-foreground">
   536→                    Adicione membros da família para começar a montar a árvore genealógica.
   537→                  </p>
   538→                </CardContent>
   539→              </Card>
   540→            ) : (
   541→              cliente.arvore.map((no) => (
   542→                <div key={no.id} className="relative group/item">
   543→                  <TreeNode 
   544→                    no={no} 
   545→                    onEdit={abrirEditarNo}
   546→                    depth={no.relacao === 'requerente' ? 0 : 
   547→                          ['pai', 'mae'].includes(no.relacao) ? 1 :
   548→                          no.relacao.includes('avo') ? 2 : 3}
   549→                  />
   550→                  
   551→                  {/* Delete button */}
   552→                  {no.relacao !== 'requerente' && (
   553→                    <button
   554→                      onClick={() => excluirNo(no.id)}
   555→                      className="absolute right-2 top-2 opacity-0 group-hover/item:opacity-100 transition-opacity p-1.5 rounded-md bg-red-50 text-red-500 hover:bg-red-100"
   556→                    >
   557→                      <Trash2 className="size-4" />
   558→                    </button>
   559→                  )}
   560→                </div>
   561→              ))
   562→            )}
   563→          </div>
   564→        </TabsContent>
   565→
   566→        {/* TAB 2: Financeiro */}
   567→        <TabsContent value="financeiro" className="mt-6 space-y-6">
   568→          <div>
   569→            <h2 className="font-serif text-lg font-600 text-navy">
   570→              Financeiro & Emolumentos Operacionais
